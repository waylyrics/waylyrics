20pub fn match_likely_lyric<'a>(
21    album: Option<&str>,
22    title: &str,
23    singer: Option<&str>,
24    length: Option<Duration>,
25    search_result: &'a [SongInfo],
26    length_toleration_ms: u128,
27) -> Option<(&'a str, u8)> {
28    length
29        .and_then(|leng| {
30            search_result
31                .iter()
32                .find(|SongInfo { length, .. }| {
33                    length.as_millis().abs_diff(leng.as_millis()) <= length_toleration_ms
34                })
35                .map(|song| (song, 1))
36        })
37        .or_else(|| {
38            // if we get only title, it is likely
39            // the player is giving messy title, which is not applicable
40            // to fuzzy-match. so we skip fuzzy-match and use
41            // the first result instead
42            if album.is_none() && singer.is_none() {
43                return None;
44            }
45
46            fn t2s(s: impl AsRef<str>) -> Vec<char> {
47                zhconv::zhconv(s.as_ref(), zhconv::Variant::ZhHans).chars().collect()
48            }
49
50            let o_title = t2s(title);
51            let o_album = album.map(t2s);
52            let o_singer = singer.map(t2s);
53
54            search_result
55                .iter()
56                .map(
57                    |s @ SongInfo {
58                         title: _title,
59                         singer: _singer,
60                         album: _album,
61                         ..
62                     }| {
63                        let r_title = t2s(_title);
64                        let r_album = _album
65                            .as_ref()
66                            .map(t2s);
67                        let r_singer = t2s(_singer);
68
69                        let likelihood = fuzzy_match_song(
70                            &o_title,
71                            o_album.as_deref(),
72                            o_singer.as_deref(),
73                            &r_title,
74                            r_album.as_deref(),
75                            &r_singer,
76                        );
77                        trace!("p={likelihood} for {s:?}");
78                        (s, likelihood)
79                    },
80                )
81                .max_by_key(|(_, likelihood)| (likelihood * 1024.) as u32)
82                .map(|(s, _)| (s, 0))
83        })
84        .or(search_result.first().map(|song| (song, 2)))
85        .map(|(song, weight)| (song.id.as_str(), weight))
86}

96pub fn fuzzy_match_song(
97    title: &[char],
98    album: Option<&[char]>,
99    singer: Option<&[char]>,
100    r_title: &[char],
101    r_album: Option<&[char]>,
102    r_singer: &[char],
103) -> f64 {
104    let title_likelihood = distance(title, r_title);
105    let singer_likelihood = || distance(singer.unwrap_or_default(), r_singer);
106    let album_likelihood = || distance(album.unwrap_or_default(), r_album.unwrap_or_default());
107    match (singer, album) {
108        (Some(_), Some(_)) => {
109            title_likelihood * 0.4 + singer_likelihood() * 0.2 + album_likelihood() * 0.4
110        }
111        (Some(_), None) => title_likelihood * 0.7 + singer_likelihood() * 0.3,
112        (None, Some(_)) => title_likelihood * 0.8 + album_likelihood() * 0.2,
113        (None, None) => title_likelihood,
114    }
115}

117pub fn extract_translated_lyric(lyric: impl AsRef<[LyricLineOwned]>) -> Vec<LyricLineOwned> {
118    lyric
119        .as_ref()
120        .windows(2)
121        .filter_map(|l| <&[LyricLineOwned; 2]>::try_from(l).ok())
122        // this should work because we have sorted lyrics by timestamp
123        .filter(|&[a, b]| a.start_time == b.start_time)
124        .map(|p| &p[1])
125        .cloned()
126        .collect()
127}

129pub fn filter_original_lyric(
130    lyric: impl AsRef<[LyricLineOwned]>,
131    tlyric: impl AsRef<[LyricLineOwned]>,
132) -> Vec<LyricLineOwned> {
133    let tlyric = tlyric
134        .as_ref()
135        .iter()
136        .map(|LyricLineOwned { text, start_time }| (start_time, text))
137        .collect::<HashMap<_, _>>();
138    let tlyric_lines = lyric
139        .as_ref()
140        .iter()
141        .filter(|LyricLineOwned { text, start_time }| {
142            !tlyric.contains_key(start_time) || tlyric[start_time] != text
143        })
144        .cloned()
145        .collect::<Vec<_>>();
146    tlyric_lines
147}
