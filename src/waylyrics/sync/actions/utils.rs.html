23#[cfg(feature = "export-lyric")]
24pub fn make_lrc_line(text: impl Display, start_time: Duration) -> String {
25    let mut ms = start_time.as_millis() as u64;
26    let mut sec = ms / 1000;
27    let min = sec / 60;
28    sec %= 60;
29    ms %= 1000;
30
31    format!("[{min:02}:{sec:02}.{ms:03}]{text}")
32}

34#[cfg(feature = "export-lyric")]
35pub async fn export_lyric(window: &Window, is_original: bool) {
36    use gtk::gio::prelude::FileExt;
37    use gtk::gio::FileCreateFlags;
38    use gtk::glib::Priority;
39    use gtk::{FileDialog, FileFilter};
40
41    info!("spawned export-lyric: original={is_original}");
42
43    let meta = TRACK_PLAYING_STATE.with_borrow(|meta| meta.metainfo.clone());
44    let current_lyrics = LYRIC.with_borrow(|l| {
45        if is_original {
46            l.origin.clone()
47        } else {
48            l.translation.clone()
49        }
50    });
51    let offset = window.imp().lyric_offset_ms.get();
52
53    let LyricOwned::LineTimestamp(lines) = current_lyrics else {
54        let error_msg = gettext("lyric not exising!");
55        error!(error_msg);
56        show_dialog(gtk::Window::NONE, &error_msg, gtk::MessageType::Error);
57        return;
58    };
59
60    let mut output = String::default();
61
62    let _ = output.write_str("[re:waylyrics]\n");
63    let _ = output.write_str(concat!("[ve:", env!("CARGO_PKG_VERSION"), "]\n"));
64    if let Some(meta) = meta {
65        if let Some(value) = meta.title {
66            let _ = output.write_fmt(format_args!("[ti:{value}]\n"));
67        }
68        if let Some(value) = meta.artists {
69            let value = value.join(", ");
70            let _ = output.write_fmt(format_args!("[ar:{value}]\n"));
71        }
72        if let Some(album) = meta.album {
73            let _ = output.write_fmt(format_args!("[al:{album}]\n"));
74        }
75    } else {
76        warn!("metainfo not found! will not generate");
77    }
78
79    let _ = output.write_fmt(format_args!("[offset:{offset}]\n"));
80    let _ = output.write_char('\n');
81
82    for line in lines {
83        output += &make_lrc_line(&line.text, line.start_time);
84        output += "\n";
85    }
86
87    let filters = gtk::gio::ListStore::new::<FileFilter>();
88    let filter = FileFilter::new();
89    filter.add_mime_type("text/plain");
90    filter.add_suffix("lrc");
91    filters.append(&filter);
92    let Ok(lrc_file) = FileDialog::builder()
93        .title(gettext("Export a lyrics file"))
94        .filters(&filters)
95        .build()
96        .save_future(Some(window))
97        .await
98    else {
99        info!("user cancelled selection");
100        return;
101    };
102
103    match lrc_file
104        .create_readwrite_future(FileCreateFlags::REPLACE_DESTINATION, Priority::DEFAULT)
105        .await
106    {
107        Ok(stream) => {
108            use gtk::gio::prelude::{IOStreamExt, OutputStreamExt, OutputStreamExtManual};
109
110            let outstream = stream.output_stream();
111            outstream
112                .write_all_future(output.into_bytes(), Priority::DEFAULT)
113                .await;
114            outstream.flush_future(Priority::DEFAULT).await;
115            outstream.close_future(Priority::DEFAULT).await;
116        }
117        Err(e) => {
118            let prompt = gettext("failed to export: ");
119            let error_msg = format!("{prompt}{e}");
120            error!(error_msg);
121            show_dialog(gtk::Window::NONE, &error_msg, gtk::MessageType::Error);
122        }
123    }
124}

126#[cfg(feature = "import-lyric")]
127pub async fn import_lyric(window: &Window, is_original: bool) {
128    use gtk::gio::prelude::{FileExt, IOStreamExt, InputStreamExt, InputStreamExtManual};
129    use gtk::glib::Priority;
130    use gtk::{FileDialog, FileFilter};
131
132    use crate::lyric_providers::utils::lrc_iter;
133    use crate::lyric_providers::Lyric;
134
135    info!("spawned import-lyric: original={is_original}");
136
137    let filters = gtk::gio::ListStore::new::<FileFilter>();
138    let filter = FileFilter::new();
139    filter.add_mime_type("text/plain");
140    filter.add_suffix("lrc");
141    filters.append(&filter);
142    let Ok(lrc_file) = FileDialog::builder()
143        .title(gettext("Select a lyrics file"))
144        .filters(&filters)
145        .build()
146        .save_future(Some(window))
147        .await
148    else {
149        info!("user cancelled selection");
150        return;
151    };
152
153    let lrc_io = match lrc_file.open_readwrite_future(Priority::DEFAULT).await {
154        Ok(lrc_io) => lrc_io,
155        Err(e) => {
156            let prompt = gettext("failed to read LRC in UTF-8: ");
157            let error_msg = format!("{prompt}{e}");
158            error!(error_msg);
159            show_dialog(gtk::Window::NONE, &error_msg, gtk::MessageType::Error);
160            return;
161        }
162    };
163
164    let inputstream = lrc_io.input_stream();
165    let lrc = {
166        match inputstream
167            .read_all_future(Vec::new(), Priority::DEFAULT)
168            .await
169        {
170            Ok((_, _, Some(e))) | Err((_, e)) => {
171                let prompt = gettext("failed to read LRC in UTF-8: ");
172                let error_msg = format!("{prompt}{e}");
173                error!(error_msg);
174                show_dialog(gtk::Window::NONE, &error_msg, gtk::MessageType::Error);
175                return;
176            }
177            Ok((bytes, _, _)) => {
178                if let Ok(lrc) = String::from_utf8(bytes) {
179                    inputstream.close_future(Priority::DEFAULT).await;
180                    lrc.to_string()
181                } else {
182                    let prompt = gettext("failed to read LRC in UTF-8: ");
183                    let error_msg = format!("{prompt}invalid encoding");
184                    error!(error_msg);
185                    show_dialog(gtk::Window::NONE, &error_msg, gtk::MessageType::Error);
186                    return;
187                }
188            }
189        }
190    };
191    let lyric = match lrc_iter(lrc.lines()) {
192        Ok(r) => r,
193        Err(e) => {
194            let prompt = gettext("input LRC in unsupported format: ");
195            let error_msg = format!("{prompt}{e}");
196            error!(error_msg);
197            show_dialog(gtk::Window::NONE, &error_msg, gtk::MessageType::Error);
198            return;
199        }
200    };
201    LYRIC.with_borrow_mut(
202        |LyricState {
203             origin,
204             translation,
205         }| {
206            if is_original {
207                *origin = Lyric::LineTimestamp(lyric).into_owned();
208            } else {
209                *translation = Lyric::LineTimestamp(lyric).into_owned()
210            }
211        },
212    );
213    let cache_lyrics = window.imp().cache_lyrics.get();
214    if cache_lyrics {
215        update_cache();
216    }
217}
