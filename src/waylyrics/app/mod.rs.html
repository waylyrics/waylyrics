41pub fn window_min_size(align: config::Align) -> (i32, i32) {
42    if align.is_vertical() {
43        (VERTICAL_MIN_WIDTH, VERTICAL_MIN_HEIGHT)
44    } else {
45        (WINDOW_MIN_WIDTH, WINDOW_MIN_HEIGHT)
46    }
47}

49pub fn build_main_window(
50    app: &Application,
51    enable_filter_regex: bool,
52    cache_lyrics: bool,
53    length_toleration_ms: u128,
54    show_default_text_on_idle: bool,
55    show_lyric_on_pause: bool,
56    respect_empty_line_as_gap: bool,
57    skip_auto_search: bool,
58    skip_taskbar: bool,
59    #[cfg(feature = "layer-shell")] layer_shell: bool,
60    #[cfg(feature = "layer-shell")] layer_shell_anchor: crate::config::LayerShellAnchor,
61) -> Window {
62    let window = Window::new(
63        app,
64        cache_lyrics,
65        length_toleration_ms,
66        show_default_text_on_idle,
67        show_lyric_on_pause,
68        respect_empty_line_as_gap,
69        skip_auto_search,
70    );
71
72    #[cfg(feature = "layer-shell")]
73    if layer_shell {
74        use gtk4_layer_shell::{is_supported, KeyboardMode, Layer, LayerShell};
75
76        if is_supported() {
77            LayerShell::init_layer_shell(&window);
78            // wlr-layer-shell-unstable-v1 version 4 needed, see
79            // https://wayland.app/protocols/wlr-layer-shell-unstable-v1#compositor-support
80            // and https://wayland.app/protocols/wlr-layer-shell-unstable-v1#zwlr_layer_surface_v1:enum:keyboard_interactivity
81            LayerShell::set_keyboard_mode(&window, KeyboardMode::OnDemand);
82            LayerShell::set_layer(&window, Layer::Overlay);
83        } else {
84            tracing::warn!("layer-shell was enabled but unsupported by the compositor!");
85        }
86
87        layer_shell_anchor.setup(&window);
88    }
89
90    let (min_width, min_height) = window_min_size(window.imp().lyric_align.get());
91    window.set_size_request(min_width, min_height);
92    window.set_title(Some(DEFAULT_TEXT));
93    window.set_icon_name(Some(crate::APP_ID_FIXED));
94
95    if skip_taskbar {
96        // the hint needs a `GdkSurface` to exist, and it has to be applied
97        // before the window is mapped, otherwise the window manager has
98        // already registered its taskbar entry
99        gtk::prelude::WidgetExt::realize(&window);
100        set_skip_taskbar(&window, true);
101    }
102
103    window.present();
104
105    let above_label = Label::builder()
106        .label("Waylyrics")
107        .name("above")
108        .vexpand(true)
109        .hexpand(false)
110        .build();
111    let below_label = Label::builder()
112        .label("")
113        .name("below")
114        .vexpand(true)
115        .hexpand(false)
116        .visible(false)
117        .build();
118
119    for label in [&above_label, &below_label] {
120        utils::setup_label(label, enable_filter_regex);
121    }
122
123    let vertical_above = VerticalLabel::new("above");
124    let vertical_below = VerticalLabel::new("below");
125    for label in [&vertical_above, &vertical_below] {
126        label.set_visible(false);
127        label.set_vexpand(true);
128        label.set_hexpand(false);
129    }
130    mirror_label(&above_label, &vertical_above);
131    mirror_label(&below_label, &vertical_below);
132
133    window
134        .imp()
135        .lyric_widgets
136        .set(LyricWidgets {
137            above: above_label.clone(),
138            below: below_label.clone(),
139            vertical_above,
140            vertical_below,
141        })
142        .expect("lyric widgets must only be initialized once");
143
144    let verical_box = gtk::Box::builder()
145        .name("lyrics-box")
146        .baseline_position(gtk::BaselinePosition::Center)
147        .orientation(gtk::Orientation::Vertical)
148        .halign(gtk::Align::Center)
149        .valign(gtk::Align::Center)
150        .vexpand(true)
151        .hexpand(false)
152        .build();
153
154    verical_box.insert_child_after(&above_label, gtk::Box::NONE);
155    verical_box.insert_child_after(&below_label, Some(&above_label));
156
157    window.set_child(Some(&verical_box));
158
159    let align = window.imp().lyric_align.get();
160    set_lyric_align(&window, align);
161
162    window.connect_decorated_notify(|window| {
163        crate::log::debug!("triggered decorated signal");
164        let clickthrough = window.imp().clickthrough.get();
165        set_click_pass_through(window, clickthrough)
166    });
167
168    window.set_icon_name(Some(crate::APP_ID_FIXED));
169
170    window
171}

173fn sync_vertical_box_empty(label: &VerticalLabel) {
174    let Some(parent) = label.parent().and_downcast::<gtk::Box>() else {
175        return;
176    };
177    if !parent.has_css_class("vertical-lyrics") {
178        return;
179    }
180    let mut has_visible = false;
181    let mut child = parent.first_child();
182    while let Some(widget) = child {
183        if widget.is_visible() {
184            has_visible = true;
185            break;
186        }
187        child = widget.next_sibling();
188    }
189    if has_visible {
190        parent.remove_css_class("empty");
191    } else {
192        parent.add_css_class("empty");
193    }
194}

196fn mirror_label(source: &Label, target: &VerticalLabel) {
197    target.set_text(&source.label());
198    target.set_visible(source.is_visible());
199
200    source.connect_label_notify(clone!(
201        #[weak]
202        target,
203        move |source| {
204            target.set_text(&source.label());
205            target.set_visible(source.is_visible());
206            sync_vertical_box_empty(&target);
207        }
208    ));
209    source.connect_visible_notify(clone!(
210        #[weak]
211        target,
212        move |source| {
213            target.set_visible(source.is_visible());
214            sync_vertical_box_empty(&target);
215        }
216    ));
217}

219pub fn set_lyric_align(window: &Window, align: config::Align) -> Option<()> {
220    let vbox: gtk::Box = window.child()?.downcast().ok()?;
221    let LyricWidgets {
222        above,
223        below,
224        vertical_above,
225        vertical_below,
226    } = window.imp().lyric_widgets.get()?;
227
228    crate::log::debug!(
229        "set_lyric_align: {align:?}, default={:?}, current={}x{}",
230        window.default_size(),
231        window.width(),
232        window.height()
233    );
234
235    for widget in [
236        above.clone().upcast::<gtk::Widget>(),
237        below.clone().upcast::<gtk::Widget>(),
238        vertical_above.clone().upcast::<gtk::Widget>(),
239        vertical_below.clone().upcast::<gtk::Widget>(),
240    ] {
241        if widget.parent().is_some() {
242            widget.unparent();
243        }
244    }
245
246    if align.is_vertical() {
247        let rtl = align.is_rtl();
248
249        vbox.set_orientation(gtk::Orientation::Horizontal);
250        vbox.add_css_class("vertical-lyrics");
251        vbox.set_halign(if rtl {
252            gtk::Align::End
253        } else {
254            gtk::Align::Start
255        });
256        vbox.set_valign(gtk::Align::Center);
257        vbox.set_spacing(4);
258
259        for label in [vertical_above, vertical_below] {
260            label.set_rtl(rtl);
261            label.set_vexpand(false);
262            label.set_valign(gtk::Align::Center);
263            label.set_hexpand(false);
264        }
265
266        vertical_above.set_text(&above.label());
267        vertical_above.set_visible(above.is_visible());
268        vertical_below.set_text(&below.label());
269        vertical_below.set_visible(below.is_visible());
270
271        if rtl {
272            vbox.append(vertical_below);
273            vbox.append(vertical_above);
274        } else {
275            vbox.append(vertical_above);
276            vbox.append(vertical_below);
277        }
278        if vertical_above.is_visible() || vertical_below.is_visible() {
279            vbox.remove_css_class("empty");
280        } else {
281            vbox.add_css_class("empty");
282        }
283    } else {
284        vbox.remove_css_class("vertical-lyrics");
285        vbox.remove_css_class("empty");
286        vbox.set_orientation(gtk::Orientation::Vertical);
287        vbox.set_halign(align.into());
288        vbox.set_valign(gtk::Align::Center);
289        vbox.set_spacing(0);
290
291        for label in [above, below] {
292            label.set_halign(align.into());
293            label.set_justify(align.into());
294            label.set_vexpand(true);
295            label.set_hexpand(false);
296        }
297
298        vbox.append(above);
299        vbox.append(below);
300        vertical_above.set_visible(false);
301        vertical_below.set_visible(false);
302    }
303
304    let (min_width, min_height) = window_min_size(align);
305    window.set_size_request(min_width, min_height);
306
307    window.imp().lyric_align.set(align);
308    window.queue_resize();
309    Some(())
310}

312pub fn get_label(window: &Window, position: &str) -> Label {
313    let widgets = window
314        .imp()
315        .lyric_widgets
316        .get()
317        .expect("lyric widgets must be initialized");
318    match position {
319        "above" => widgets.above.clone(),
320        "below" => widgets.below.clone(),
321        _ => panic!("unknown lyric position: {position}"),
322    }
